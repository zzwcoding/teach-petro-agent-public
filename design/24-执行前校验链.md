### 主干一：GuardChain.validate——Verdict 三态、五段顺序与短路序（链本体）

链的全部机制摆在明面上：一个段序数组、一次循环、一处短路 `break`、一个出口审计。没有插件、没有中间件、没有管道抽象——**确定性检查就该一眼看穿**（锚 SchemaFlow）。

```text
伪代码 — app/agents/text2sql/guardchain/chain.py:GuardChain（m17；签名=卡 m17）

class Verdict:                  # 数据形状——链的唯一出口契约（卡面：allow(annotated) | reject(stage, reason, fix_hint)）
    status: str                 # "allow" | "reject"
    sql: str | None             # allow 时：改写后规范形（AST 序列化+策略注入）——原样语句不出现在这个字段里
    annotations: list[str]      # 注记位（空列表=干净放行）：stale 过期警示/新鲜度 unknown 表/方言近似（设计稿 26 起）
    explanation: Explanation | None   # 血缘段解释链（五要素，主干六）
    stage: str | None           # reject 时：ast | policy | freshness
    reason: str | None          # reject 时：结构化码——parse_fail / multi_statement / readonly_violation /
                                #   policy_table_denied / policy_column_denied / freshness_over_window
    fix_hint: dict | None       # reject 时：{stage: 哪段, rule: 什么规则, fix: 怎么改}——取舍 4

class GuardChain:
    def __init__(self, ast段, policy段, freshness段, lineage段, audit段):
        # 五段构造注入（设计稿 13 boot 装配同款时机，本设计稿不写装配代码）；段序在此锁死——
        # 语法→策略→新鲜度→血缘，调用方（m7）连段序都看不见，更改不了（卡 m17 Seam："五段顺序与短路语义锁死本卡"）
        self._段序 = (ast段, policy段, freshness段, lineage段)   # 前四段参与短路
        self._audit = audit段                                    # 审计段=链出口，不参与短路（取舍 3）

    async def validate(self, sql, user, metric_ref=None) -> Verdict:
        # ── 主干判断① 段一：语法闸（stub 替身段——设计稿 15 三判，设计稿 26 换真身）──
        解析 = self._段序[0].parse(sql)                    # -> 解析产物 | ParseError（主干三）
        拒绝 = None
        if isinstance(解析, ParseError):                   # 解析失败/多语句/DML——reason 码区分三型
            拒绝 = Verdict(status="reject", stage="ast", reason=解析.reason,
                           fix_hint=语法闸改法(解析.reason))
        # ── 主干判断② 段二：策略闸（注入改写或权限拒——异常去向：权限拒→理由入轨迹，通用约定）──
        if 拒绝 is None:
            try:
                改写形, 谓词摘要 = self._段序[1].inject_filters(解析.规范形, user)   # 主干四
            except PolicyDenied as d:                      # 整表越权/列越权——结构化拒绝载荷
                拒绝 = Verdict(status="reject", stage="policy", reason=d.reason, fix_hint=d.fix_hint)
        # ── 主干判断③ 段三：新鲜度闸（唯一 IO 判——水位读 m3 通用快照面/FakeFreshness）──
        注记, 解释 = [], None
        if 拒绝 is None:
            新鲜 = await self._段序[2].check(解析.引用表)   # -> fresh | stale(batch_ts, 落后, 超窗) | unknown（主干五）
            if 新鲜.超窗:
                拒绝 = Verdict(status="reject", stage="freshness", reason="freshness_over_window",
                               fix_hint={"stage": "freshness",
                                         "rule": f"承诺水位 {新鲜.承诺水位}，实际批次 {新鲜.batch_ts}",
                                         "fix": "等待批次刷新后重试，或要数据 owner 明确放宽水位承诺"})
            else:
                if 新鲜 is stale:                          # 窗内落后——不静默：标注放行（取舍 2 的另一支）
                    注记.append(f"数据截至 {新鲜.batch_ts} 批次（承诺 {新鲜.承诺水位}）")
                if 新鲜.缺登表:                            # unknown 分支——跳过比对但留痕（取舍 2）
                    注记.append(f"新鲜度未知（无水位登记）：{'、'.join(新鲜.缺登表)}")
        # ── 主干判断④ 段四：血缘标注（纯标注从不拒绝；metric_ref 可空 → 降级解释链）──
        if 拒绝 is None:
            解释 = self._段序[3].annotate(metric_ref, 解析.引用表)      # 主干六（前四要素）
            解释.过滤条件 = 谓词摘要 + 原有条件列清单(解析.树)            # 链装订第五要素——
            #   annotate 签名只收 (metric_ref, tables)，注入谓词只有链见过：装订收在汇合处，不造第二通道
        # ── 短路落点：拒绝非空时其后所有段未执行——短路序=段序的前缀 ──
        verdict = 拒绝 or Verdict(status="allow", sql=改写形,
                                  annotations=注记, explanation=解释)
        # ── 主干判断⑤ 段五：审计出口必达（含短路 reject——否则拒绝恰是唯一不留痕的路径，取舍 3）──
        await self._audit.log(verdict, user)
        return verdict
        # 异常去向三层：段内可纠正拒绝→折 Verdict.reject 由 check_sql 映射；新鲜度源不可达→结构化上抛
        #   （装配/基础设施级故障，非查询错误——折错误帧归设计稿 19，不走反思回边）；本链零模型调用、
        #   零拒绝话术生成（话术是 m7 的固定文本）——确定性检查不碰模型（锚 SchemaFlow）
```

**Verdict 三态的说法**：卡面写 `allow(annotated) | reject(...)`——"三态"实现为 **allow 干净 / allow 带注记 / reject**：注记位非空与否是调用方可辨的第三态（stale 警示、unknown 表、方言近似统一走这里），不设第三个 status 值。理由在取舍 3。

```text

class PolicyDenied(Exception):  # 结构化拒绝载荷（异常去向"权限拒→理由入轨迹"的载体）
    reason: str                 # policy_table_denied | policy_column_denied
    fix_hint: dict              # {stage, rule, fix} 三元组——取舍 4

class PolicyStage:
    def __init__(self, 策略表):
        # 表→（行权限谓词模板, 列可见集），角色→可读表集：部署期登记的配置形状注入——
        # m17 依赖行没有权限系统对接项，权限数据以配置进链（取舍 1 的边界前提）；走查=固定配置
        self._策略表 = 策略表

    def inject_filters(self, sql, user) -> (sql', 谓词摘要):
        # 输入=段一规范形（单条 SELECT、已收口 LIMIT）；再解析一次零风险（同款调用形状同方言）
        树 = sqlglot.parse(sql, read="postgres")[0]
        # ── 主干判断① 整表越权：引用表 ∉ 该角色可读表集 → 权限拒（W-F14-1 不适用——那是缺过滤，这是无权）──
        越权表 = {t.name for t in 树.find_all(exp.Table)} - self._策略表.可读表集(user.role)
        if 越权表:
            raise PolicyDenied("policy_table_denied",
                               fix_hint={"stage": "policy",
                                         "rule": f"角色 {user.role} 无权读取：{'、'.join(越权表)}",
                                         "fix": "去掉对无权表的引用重问，或向数据 owner 申请授权"})
        # ── 主干判断② 列越权：查询要用的列 ∉ 该角色可见列 → 权限拒（列裁剪救不了查询本身要的列）──
        for t in 树.find_all(exp.Column):
            if not self._策略表.列可见(user.role, t.table, t.name):
                raise PolicyDenied("policy_column_denied", fix_hint=列改法(t))
        # ── 主干判断③ 行权限 mandatory filter 注入：AST AND 复合，禁字符串拼接（设计稿 15 取舍 3 同款纪律）──
        谓词 = self._策略表.行权限谓词(user.role, 引用表(树))   # 模板按用户权限实例化，
                                                               # 如 "w.block_name IN ('渤海油田')"
        if 谓词 is not None:                # None=该角色全域可见——注入零成本跳过，不是漏注入
            树 = 树.where(谓词)             # 已有 WHERE 也追加复合——mandatory 语义=多滤不少滤；
                                            # 不检测"语义已覆盖"（确定性优先，走查推演 3 有实例）
        return 树.sql(), 谓词摘要(谓词)     # 数据变形：AST 序列化=改写后规范形——执行/指纹/审计三处同源
        # 异常去向：PolicyDenied 折进链的 reject verdict（主干一判断②）；本段零 IO 零模型调用
```

### 主干五：FreshnessStage.check——批次水位比对（W-F14-2）

```text
伪代码 — app/agents/text2sql/guardchain/stages/freshness.py:FreshnessStage（m17；签名=卡 m17）

class 水位结论:                 # 数据形状——三态的统一载荷（卡面：fresh | stale(batch_ts) | unknown）
    态: str                     # fresh | stale | unknown
    batch_ts: str | None        # 最旧引用表的实际批次时间（stale 时）
    落后: str | None            # 相对承诺水位的落后量（如 "2 天"）
    超窗: bool                  # 落后是否超出容忍窗——链据此分档：False=标注放行，True=reject
    承诺水位: str | None        # 登记行原文（如 "T-1 每日批次"）——fix_hint 与标注的原材料
    缺登表: list[str]           # 无水位登记的表（unknown 注记的原材料）

class FreshnessStage:
    def __init__(self, 水位源):
        # 生产实现=m3 通用快照面 get_snapshot("freshness:{表名}")（语义扩展备案——越界自查）；
        # 走查替身=卡面 Adapter FakeFreshness（固定批次水位）
        self._水位源 = 水位源

    async def check(self, tables) -> 水位结论:
        # ── 主干判断：全表取最旧——一表陈旧即整查询陈旧（JOIN 出来的行集不会比最旧的表更新）──
        最旧落后, 最旧批次, 承诺, 缺登 = None, None, None, []
        for t in sorted(tables):                    # 排序遍历：水位行读取序确定，审计可复现
            行 = await self._水位源.水位(t)          # {"batch_ts", "承诺水位", "批次周期"} 或 None
            if 行 is None:
                缺登.append(t)                      # unknown 原料——取舍 2：跳过并标注，不拦截
                continue
            落后 = 承诺批次日(行) - 实际批次日(行)    # 数据变形：日历差换算（如承诺 T-1、实际 T-3 → 2 天）
            if 最旧落后 is None or 落后 > 最旧落后:
                最旧落后, 最旧批次, 承诺 = 落后, 行["batch_ts"], 行["承诺水位"]
        if 最旧落后 is None:                        # 引用表全无登记
            return 水位结论(态="unknown", 缺登表=缺登)
        容忍窗 = 批次周期(承诺)                      # 取舍 2：容忍窗=一个批次周期（登记行自带，不硬编码）
        态 = "fresh" if 最旧落后 <= 0 else "stale"
        return 水位结论(态=态, batch_ts=最旧批次, 落后=f"{最旧落后} 天", 超窗=(最旧落后 > 容忍窗),
                        承诺水位=承诺, 缺登表=缺登)
        # 异常去向：水位源不可达 → 结构化上抛（链不吞基础设施故障——装配/库级问题折错误帧归设计稿 19）；
        # 单表无登记不是故障 → unknown 注记（取舍 2）
```

## 关键判断的取舍记录

1. **PolicyStage 的注入=SQL 改写，不是拒绝重生成；越权才是拒绝。** 备选是"发现缺过滤就 reject 回模型重写"——被演讲轮 5 的机制观察否掉：mandatory filter 缺失的根因是"注意力窗口机制导致过滤条件漏掉"（前沿演讲），重生成赌的是模型这次记得，换一次采样而已，还多烧一轮 token；闸替模型补（`树.where(谓词)` AST 复合）是确定性改写，一次到位零往返——这正是"错误信息该指路"原则的更进一步：能代写的直接代写。注入不是无代价：改写后的语句与模型原意有差（多了一个它不知道的过滤），所以解释链把注入谓词写进过滤条件（走查推演 3 t2），差是显性的。**切换条件**：注入从"补谓词"升级为"重排查询结构"（如权限要求改写 JOIN 对象）时，改写就变成了另一种生成——那时才切换为拒绝重生成，并把权限上下文喂进生成提示词。越权表/列不注入（想注入也没地方放），走拒绝且不回灌——权限缺口不是模型的错，回灌纠正不了资格。
2. **新鲜度水位来源=m3 通用快照面的登记行；unknown 跳过并标注；容忍窗=一个批次周期。** 水位登记行（batch_ts/承诺水位/批次周期）由批次落地侧写入 m3 快照面（`freshness:{表}` 键），链只读不写——登记义务在数据侧（设计范围外边界），走查用卡面 Adapter FakeFreshness 固定水位，生产换真不动段签名。unknown（无登记表）跳过比对并标注而不拦截：登记覆盖是渐进的治理过程，链一步到位全拦会把"登记没铺完"变成"问数全停"——治理进度惩罚业务，本末倒置；但跳过必须留注记（State.explanation 可见），透明替代静默。容忍窗=一个批次周期且随登记行走（不硬编码）：日批表容忍 1 天、月批表容忍 1 月，同一判据适配两种节奏。**换边条件**：出现"水位承诺 SLA + 破窗告警"的治理需求时，登记行升级为带 owner 的水位合同（漂移告警归设计稿 23 评估链的回归半边，不进本链）。
3. **Verdict 的第三态放注记位，不设第三状态值；短路序锁前四段，审计段是出口不是短路对象。** 卡面 `allow(annotated) | reject` 只有两个 status 值；stale 警示、unknown 表、（设计稿 26 的）方言近似都需要"过了但带话"的第三种可观察形态——设第三个 status 值会让 check_sql 与未来每个消费方都多一条分支，而注记位（`annotations` 非空与否）零分支成本即可辨。短路序：前四段任一 REJECT 即停其后所有段（`freshness` 拒时血缘段不跑——等不到新鲜数据，出解释链没有意义）；但审计段在链出口必达，短路不短审计——W-F14-5 写的是"任意 verdict"，而拒绝恰是最需要追责的路径，若审计参与短路，链就有"不留痕的裁决"，可追溯性在最要紧处失明（走查推演 2 t3）。**State 备案**（取舍 3 尾注）：注记与解释链从校验节点流到结果校验节点，图内唯一合法通道是 State——增 `explanation` 可选格，随节备案在模块卡 State 行补记（设计稿 22 metric 格"图内唯一合法通道是 State"同款论证）。
4. **fix_hint 定死"可执行错误信息"三元组：{stage: 哪段, rule: 什么规则, fix: 怎么改}。** 锚 SchemaFlow："把错误响应 prompt 工程化，清楚传达具体可执行的改进建议"，而非不透明错误码或堆栈——回灌场景里 fix_hint 直接顶设计稿 14 error 结构的 hint 格（主干二回灌出口），模型拿到的是"怎么改"不是"你错了"；出图场景里 fix_hint 原文进拒绝话术，老王拿到的是"等批次刷新"不是"校验失败"。stage 字段让同一格式的 hint 在回灌与出图两条路通用；rule 字段留住规则原文供审计对账（审计行的 fix_hint 与话术逐字同源）。**边界**：fix_hint 只指路不代写——越权与旧批次不存在"模型能改对"的修法（资格与数据问题），这两类的 fix 指向人（申请授权/等批次），只有 readonly_violation 指向模型重写——错误信息"可执行"的前提是收件人真的执行得了。

## 锚点对照精要

**SchemaFlow 阶段间确定性检查**：原文——工作流把任务拆成显式阶段，"Adds deterministic checks between stages so obvious failures are caught before downstream steps consume bad state"，且"不能替代人工评审，但能尽早抓住常见的静默失败"。落点：GuardChain 五段就是生成之后、执行之前的那排阶段间确定性检查——Verdict/解析产物/水位结论是阶段间契约，坏状态（无 LIMIT、缺过滤、旧批次）过不了段界就到不了执行器。差异：SchemaFlow 的检查站在"生成各阶段之间"，我们收拢成执行前一道链；它声明不对真实库执行，我们链后还有真实执行与设计稿 15 人工审批位——"确定性检查不替代人审"由审批位续接。

**可执行错误信息**：原文——"prompt-engineer your error responses to clearly communicate specific and actionable improvements"，截断提示要引导 agent 用过滤/分页。落点：fix_hint 三元组（取舍 4）与回灌 hint 对接（主干二）——被拦的 SQL 拿到"哪段/什么规则/怎么改"；LIMIT 注入本身就是"引导用分页"的闸版。差异：Anthropic 说的是工具返回给 agent 的报错，我们把同一纪律用到两个收件人——回灌给模型（hint）、出图给人（话术），收件人不同、三元组同构。

**权限穿透（pass-through permissions）**：原文——OpenAI 内部数据智能体"Users only query tables they already have permission to access"，越权时标记缺失访问或降级回退。落点：策略段的可读表集/列可见集判+越权拒绝（主干四）是同一原则的应用侧实现。差异：OpenAI 靠连接账户权限让库自己做 enforcement（库是闸），我们把行列权限做进执行路径的 SQL 本体（mandatory filter 注入）+ 策略段先拦——因为"能跑但缺过滤"的查询在库侧权限里是合法的，应用层必须补库权限管不到的那半边（前沿演讲指出的原始问题）。

**失败阈值人机介入**：原文——人机介入两类触发器："Exceeding failure thresholds: Set limits on agent retries or actions"、"High-risk actions… should trigger human oversight"。落点：本链的拒绝仍走设计稿 14/15 既有 attempts 熔断预算（回灌分支上限闸前置，主干二）——阈值触发转人工的机制零改动；高风险动作的人工监督仍由设计稿 15 审批位承担（对照件：有权但超常的外表查询挂起等人批）。差异：本链不新增人机介入点——它是确定性闸，拦得住的自己拦，拦不住的（资格、超常）交给既有的两类触发器，闸不抢审批位的活（捣乱输入 B t3）。

**前沿演讲：执行前校验链**：原文——SQL 有结果但业务不能直接用：mandatory filter（行列权限、必要 LIMIT——"注意力窗口机制导致过滤条件漏掉"）可能缺失；收敛"自由执行"为"先校验再执行：policy check → freshness check → lineage check → audit log → execute"。落点：五段序即其四段前插语法闸（设计稿 15 既有资产升级为首段——设计稿 00 stub 替身声明里 sql_guard 升级路线的兑现），段序一字不差沿用其 policy→freshness→lineage→audit。差异：演讲的 SQL 编译走 MetricFlow 类工具，我们的"编译"是模板参数化（设计稿 22）+ AST 注入；其"执行前校验"在我们的段序里多出审计出口必达的设计（取舍 3），是把"audit log"从并列段升为出口记录者。

**Microsoft Fabric data agent**：原文——全程用 requesting user's credentials，"least-privilege access"，RLS/CLS 依旧生效，"strictly enforces read-only access"；对话输出上限 25 行 × 25 列——"designed for conversational insights rather than for returning complete datasets"。落点：行数收口（默认 50/收口 200）是 25 行裁剪的简化演绎——商业产品直接替用户决定"大结果不进对话"，我们的闸在 SQL 本体收口（AST 注入 LIMIT）+ 工具侧 fetchmany 收口（设计稿 12）双层；"RLS/CLS 依旧生效"对应策略段与库侧权限的纵深关系——注入不替代库内行级安全，库账号只读（设计稿 15）仍兜底。差异：Fabric 靠凭证穿透让库 enforce，我们在应用层把权限写进语句——两条路线在取舍 1 的切换条件处汇合（重排级权限需求出现时向凭证穿透靠拢）。
