class MrFlow:
    def __init__(self, semantic_ci, registry, repository):
        self._ci = semantic_ci                  # m18 卡内依赖：送检只调 validate_change（设计稿 38 交付物）
        self._registry = registry               # m16 公开面：合并/回滚后 load 重载（设计稿 22 既有形状引用）
        self._repo = repository                 # m3 公开面：变更包快照+审计行（设计稿 22/38 同款）
        # 🛡️ 结构自检（红线①②③的总开关）：本类没有写 semantics/ 的文件句柄、没有绕过 approve_mr
        #    的 release 通道、没有改派生物的任何入口——三个"没有"是构造签名决定的，不是纪律提倡的

    async def propose_mr(self, diff, submitter) -> mr_id:       # 卡面签名逐字
        # ── 形状闸① 🛡️ 只改源：diff 每个路径必须在权威源四目录白名单内（红线①的结构半边）──
        if 有路径出白名单(diff):
            raise DiffRejected("变更包只收权威源文件——派生物由装载重建，不接受文件变更（红线③）")
        # ── 形状闸② 最小 diff：一处根因解释所有文件——"一次反馈=一处根因=一个 MR"的机械判据 ──
        if not 根因定位(diff) or len(根因集合(diff)) != 1:
            raise DiffRejected("根因定位必填且唯一——多根因拆包提审：一个 MR 只背一处根因，"
                               "混合变更让回滚与追责都失去对象（取舍 1）")
        for 文件 in diff:
            if not 被根因解释(文件, 根因定位(diff)):
                raise DiffRejected(f"{文件.路径} 与根因无关——'顺手改'拆出独立 MR（最小 diff 判据）")
        # ── 形状闸③ 建议来源可查：投向核对（设计稿 42 衔接）──
        if 建议来源.类型 == "suggestion":
            建议 = await 回读建议(建议来源.号)              # 只读设计稿 42 落库快照（不采不诊——设计稿 42 地盘）
            if 建议.投向 != "mr" or 建议.状态 not in ("triaged", "proposed"):
                raise DiffRejected("该建议未经分诊确认投向 mr——池子只产建议，提案要过确认（设计稿 42 状态机）")
        # ── 登记：状态 in_review，落库+审计（ propose 不跑 CI——提交便宜，裁决在合并闸前，主干三）──
        mr_id = "mr-" + 日期序号()
        await self._repo.save_snapshot(f"mr:{mr_id}", 序列化(变更包))
        await self._repo.save_audit({"类型": "MR 登记", "幂等键": mr_id, "提交者": submitter,
                                     "根因": 根因定位(diff), "建议来源": 建议来源})
        return mr_id
        # 异常去向：DiffRejected 是业务闸不是故障——修法随异常文本给到提交侧（agent:triage 会把
        #   修法回写建议池分诊注记；human 提交人直接改包重提）；m3 不可达结构化上抛（旁路不拦问数）
        # ★ 本函数的"不自行合并、不碰派生物"（W-F25-1 断言半边）由返回值类型背书：
        #   propose_mr 的返回只有 mr_id——一个提案号，不是生效，不是文件写入，不是发布

**"最小 diff"为什么值得一道闸**：diff 的诱惑是"都来了，顺手把相关的也改了"。但变更包的每一份文件都要过五级安检、都要在回滚时整体回退、都要在评审会上逐行过目——捆绑变更把三件事的成本全部乘上文件数，而其中往往只有一个文件与根因有关。"一处根因解释所有文件"把"最小"从倡导变成判据：解释不了的文件进不了包。锚 DataBuddy 受控进化（AE 前沿方案报告 §⑤ 受控进化行）原话"一次反馈 = 一处根因 = 一个 MR"——这句话的机制含义就是本闸：反馈（sg-0003）是计数单位，根因（缺别名）是诊断单位，MR（一行增补）是变更单位，三个单位一一对齐，回滚与追责才有不可辩驳的对象。

### 主干三：approve_mr 审批闸与合并发布——release 的唯一出口（W-F25-2 的本体）

```text
伪代码 — app/eval/mr_flow.py:MrFlow.approve_mr（m18；签名=卡 m18 版本 3 扩容节逐字；
         🛡️ 合并 = SemanticCI 全绿 + Owner 审批（卡面注释逐字）——双前置缺一不放行）

    async def approve_mr(self, mr_id, reviewer) -> release:      # 审批动作本身触达合并
        包 = await self._repo.get_snapshot(f"mr:{mr_id}")
        if 包 is None or 包.状态 != "in_review":
            raise MrNotInReview(mr_id)              # 幂等闸：重复审批/已合并/已驳回都不再触达
        # ── 前置① 送检裁决：五级 gate_report 全绿（设计稿 38 只调用——本函数不重写任何判据）──
        report = await self._ci.validate_change(包)
        if 任一红(report):
            await self._repo.save_audit({"类型": "MR 打回", "幂等键": mr_id, "红项数": 数红(report)})
            raise MrGateRejected(mr_id, report)     # 报告即修法清单（设计稿 38 四元组：级/项/结论/修法）
        # ── 前置② 审批资格：审批证据与影响档匹配（设定书 §1 责任链路由）──
        if 包.影响分析.分类 == "高" and not 审批匹配(包.审批证据, 影响档="高"):
            raise MrApprovalMissing(mr_id, 审批人路由(包))    # W-F25-2：无审批不放行（走查推演 2）
        #   审批匹配判据：高影响档=数据治理领导组（设定书 §1 角色卡"高影响变更审批、Review Board"）
        #   + 变更对象 Owner 联席（产量口径=开发室，设定书 §1/§3）；低影响档=Owner 例行签认
        #   （设计稿 38 主干五注沿用：免批先过 CI，放行仍须签认——红线②的地盘，见四红线段）
        # ── 合并发布：版本指针推进 + 写回重载（m16 生命周期既有形状，设计稿 22 引用不复写）──
        release = await self._合并发布(包, reviewer)
        #   = 权威源文件写回（diff 应用）→ self._registry.load(权威目录) 重载生效 →
        #     release{release_id, mr_id, 版本指针: 旧→新, 生效清单=manifest 快照版本}
        await self._repo.save_audit({"类型": "MR 合并", "幂等键": f"{mr_id}:{release.release_id}",
                                     "reviewer": reviewer, "版本": release.版本指针.新})
        if 包.建议来源.类型 == "suggestion":
            await 回写终态(包.建议来源.号, "merged",                 # 设计稿 42 备案公开面——设计稿 42 钩子兑现：
                           决议记录=f"{mr_id} 由 {reviewer} 审批合并")  #   状态机后半段的触达者=本节
        return release
        # ★ release 的唯一出口就是本函数：Agent 侧没有任何可单方触达 release 的路径——
        #   "只提 MR 不自行合并"在接口上不可表达（红线②的结构半边）
        # 异常去向：MrGateRejected/MrApprovalMissing 是业务闸（修法指名收件人，设计稿 38 四元组纪律）；
        #   m16 写回/重载故障结构化上抛——合并是事务语义：写回失败不留半包（回滚指针即可，主干四）
```

### 主干四：回滚语义——版本指针回拨 vs 派生物重建（W-F25-3 的本体）

```text
伪代码 — app/eval/mr_flow.py:MrFlow.rollback（m18 备案件；授权出处=设计范围伪代码梳理范围
         "回滚语义：版本指针回拨 vs 派生物重建"行 + 卡面注释"版本发布可回滚"）

    async def rollback(self, release_id, operator, 理由) -> release:
        rel = await self._repo.get_snapshot(f"release:{release_id}")
        # ── 回滚也是治理决议：触发须带决议依据（高影响回滚=领导组决议，设定书 §1；低影响=Owner）──
        #    时钟/机器不代行废弃决议（设计稿 35 取舍 2 同款哲学：治理动作不可静默代行）
        # ── 半边一 版本指针回拨（管"源回到哪一版"）：权威源文件回到该 release 之前的版本 ──
        #    概念 Git 演绎：权威源每次合并即存快照（commit 概念），回拨=指针指回上一快照——
        #    一份反向 diff 一次应用，绝不逐处手捞（手捞=把回滚变成一场新的直改，红线①还魂）
        await 权威源回拨(rel.版本指针.旧)
        # ── 半边二 派生物重建（管"投影如何追上源"）：registry.load 全量重载 ──
        await self._registry.load(权威目录)      # 倒排/manifest/模板投影全部重算——重建，不是回滚派生物
        await self._repo.save_audit({"类型": "MR 回滚", "幂等键": release_id, "操作者": operator,
                                     "理由": 理由, "回拨至": rel.版本指针.旧})
        return 回滚记录(release_id, rel.版本指针.旧)
        # ★ 分工线一句话：回拨管源、重建管影——回滚=回拨+重建，次序不可反，对象不可换：
        #   只重建不回拨=源没回去，重载只是把新口径再装载一遍；只回拨不重建=线上投影仍是新口径
        #   （内存态不随文件自动变），"回滚成功"是假象。投影永远没有独立意志（取舍 2）
```

## 关键判断的取舍记录

1. **"一个 MR 只背一处根因"是形状闸判据，不是倡议。** 备选：允许关联变更捆绑提审（"反正一个评审会"）。否决理由：捆绑变更让回滚失去对象（退 A 必连带退没病的 B）、让评审失焦（逐行过目变成抽签）、让追责失锚（根因定位字段写成"多处待查"等于没写）；锚方案"一次反馈 = 一处根因 = 一个 MR"是计数单位对齐，不是修辞。落定：根因定位必填唯一、每个 diff 文件须被根因解释（主干一形状闸②），解释不了的拆包。代价：相关变更要走多次全流程，慢。收益：每次变更可独立验证、独立回滚、独立追责。换边条件：无（本节灵魂）。
2. **投影零独立意志：回滚=回拨（源）+重建（影），派生物永不"回滚"。** 备选一：回滚时把派生物也逐个恢复旧值。否决理由：派生物是重算产物不是状态积累，"恢复旧值"等于承认投影有独立历史——与"引用结构上不存在旧副本"的一致性论证（设计稿 22 走查推演 4 t2）直接矛盾，且逐个恢复必然漏（红线③的欠债清算）。备选二：不提供回滚，"再提一个反向 MR"。否决理由：反向 MR 适用于"口径新决议"（往前进），不适用于"发布错了"（往回退）——后者要的是恢复已知好状态，指针回拨是它的最小操作；且反向 MR 也要走五级+审批，止血慢一轮。落定：回拨+重建两半边，次序不可反（主干四注）。换边条件：发布规模大到全量重载成为瓶颈——影子上先切、源后追平（Shadow Build 原子切流方向，设计稿 38 概念段），本设计稿不预支。
3. **propose 不跑 CI，裁决集中在合并闸前一次。** 备选：propose_mr 即触发送检（提交即验证）。否决理由：送检有成本（L5 全量跑批是真金白银——设计稿 38 主干一产能论证），而相当比例的包会死在"根因定位不合格/来源不可查"的形状闸上——把贵验证垫在便宜形状闸之后是既定哲学（设计稿 38"贵闸永远最后"同构）；且送检报告挂在 approve 时点，评审会看的是同一份最新裁决，不存"提交时的旧报告"。落定：propose=形状闸+登记；validate_change 在 approve_mr 前置①调用。换边条件：变更包积压到评审会过载——propose 时做廉价预检（L1–L4 静态半边），L5 仍留合并闸前，判定序不变。
4. **四红线写进结构而不是写进提示词。** 备选：在 Agent 系统提示词里写"请遵守四条红线"。否决理由：提示词是上下文层护栏，可被注入顶掉（设计稿 15"代码即规则"的同款论证：提示词写着只能 SELECT 模型仍给 DELETE）——红线的对面是一个可能被诱导、可能犯错的 Agent，靠自觉的防线对它无效。落定：①=diff 路径白名单、②=release 单出口+reviewer 必填、③=投影零写出口、④=审批档匹配——四条全是构造签名与接口形状（主干一/三/四），Agent 遵守红线不需要美德，只需要没有那条路。换边条件：无。
5. **回滚触发也是治理决议，不设自动回滚。** 备选：CI/监控发现异常自动触发 rollback（self-healing）。否决理由：异常判据本身是口径问题（"差额多大算异常"——设定书 §4 规则 4 容差族是登记出来的治理决议）；自动回滚会在"指标抖动"与"口径错误"分不清的灰区里替 Owner 做废弃级决议（设计稿 35 取舍 2"时钟不替负责人做废弃决议"同哲学）；且回滚改写权威源指针，与直改主干在权力等级上是同一件事——它的输入也该过闸。落定：rollback 带 operator+理由+决议依据，审计留痕；自动化止于提案的对偶是"自动化亦止于回退"。换边条件：出现"回滚判据登记表+值守 Owner 常驻"的运营档位后，低影响档回滚可预授权（仍留痕）。
