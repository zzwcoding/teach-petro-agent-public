### 主干一：make_ppt——分析结果 JSON 到模板成品（W-F10-1 生成半程；卡 `m10` 公开接口）

```text
伪代码 — app/multimodal/ppt_gen.py:make_ppt（m10；签名=00 号骨架登记 m10 第一行）

async def make_ppt(analysis_json, template) -> PptFile:
    # ── 主干判断① 模板存在性（快速失败：没有模板就没有布局依据，拒绝无模板出片）──
    if not 模板存在(template):
        raise 生成输入不合(f"模板文件缺失：{template}——演示文稿生成拒绝无模板出片")
    # ── 主干判断② 分析结果完整性：结论要点是版式的最低进料，缺了封面之外无页可装 ──
    if not analysis_json.get("结论要点"):
        raise 生成输入不合("分析结果缺结论要点——先回问数链补分析再出片")
    # ── 数据变形① 分析 JSON → 模板布局参数（确定性映射，主干零模型调用——进阶见主干二）──
    版式索引 = 读模板版式表(template)
    #   ★ 官方示例 [0]=标题页 / [1]=标题+项目符号 / [5]=仅标题 / [6]=空白——
    #     索引因模板而异，按模板配置查表取用，不在代码里硬编码数字
    布局参数 = {"封面":  {"标题": analysis_json["标题"],
                         "副标题": f"{analysis_json['快照来源']} · {今天}"},
                "结论":  {"标题": "结论与建议", "要点列表": analysis_json["结论要点"]},
                "图表":  {"标题": "产量预测走势", "数值表": analysis_json["图表数据"],
                         "图件": analysis_json.get("图件引用")}}   # 图件可选——miss 走降级
    # ── 数据变形② python-pptx 官方形状生成主干（设计范围红线调用序列逐条落；
    #    python-pptx 是纯本地 CPU 操作、零网络 IO，事件循环驻留毫秒级）──
    prs = Presentation(template)                     # ① 基于模板打开（非空白 Presentation()）
    # 封面页：add_slide + shapes.title（占位符快捷访问）
    slide = prs.slides.add_slide(prs.slide_layouts[版式索引["标题页"]])
    slide.shapes.title.text = 布局参数["封面"]["标题"]
    slide.placeholders[版式索引["副标题位"]].text = 布局参数["封面"]["副标题"]
    #   ★ 占位符索引同版式索引一样按模板查；并非所有 shape 都能容纳文本（官方：
    #     has_text_frame 可判断）——查表配置只指向确有文本框的占位符
    # 结论页：标题+项目符号版式，逐段写 bullet
    slide = prs.slides.add_slide(prs.slide_layouts[版式索引["标题加项目符号"]])
    slide.shapes.title.text = 布局参数["结论"]["标题"]
    正文框 = slide.placeholders[版式索引["正文位"]].text_frame
    首段 = True
    for 要点 in 布局参数["结论"]["要点列表"]:
        p = 正文框.paragraphs[0] if 首段 else 正文框.add_paragraph()
        # ★ 官方形状坑：text_frame.text 整体写入会清空除首段外的所有段落——逐段 add_paragraph
        p.text = 要点
        p.font.size = Pt(20)
        首段 = False
    # 图表页：仅标题版式 + 数值表 + 可选插图
    slide = prs.slides.add_slide(prs.slide_layouts[版式索引["仅标题"]])
    slide.shapes.title.text = 布局参数["图表"]["标题"]
    表 = slide.shapes.add_table(rows=len(布局参数["图表"]["数值表"]["日期"]) + 1, cols=2,
                                Inches(1), Inches(2), Inches(8), Inches(3)).table
    表头与数据逐格写入(表, 布局参数["图表"]["数值表"])   # 关键数值逐格相等——可断言的产物（取舍 1）
    if 布局参数["图表"]["图件"] is not None:
        图件路径 = await 解析图件引用(布局参数["图表"]["图件"])    # 按引用取图件工件（IO await）
        if 图件路径 is None:
            # ── 异常去向① 图件 miss：占位文本降级——页照出、缺局可见，不打断整份产物 ──
            提示框 = slide.shapes.add_textbox(Inches(1), Inches(6), Inches(8), Inches(1))
            提示框.text_frame.text = f"图件缺失：{布局参数['图表']['图件']}（引用失效，占位标示）"
        else:
            slide.shapes.add_picture(图件路径, Inches(1), Inches(6), height=Inches(2))
    # ── 数据变形③ 保存为产物候选（生成与交付分离：只出候选，落桶落库归 worker）──
    临时文件 = 临时目录 / f"ppt-{产物号}.pptx"
    prs.save(临时文件)                                # ② 官方保存形状（设计范围红线末环）
    return {"流": 打开(临时文件, "rb"), "扩展名": ".pptx", "页数": len(prs.slides)}
    # PptFile=产物候选形状（卡面 PptFile 的展开）：流与 m15 落桶 put_stream(bucket, key, stream) 对接
    # 异常去向②：模板损坏/版式索引越界 → python-pptx 异常 → 包装 生成失败 上抛 → 设计稿 21 失败路径
```

```text
伪代码 — app/multimodal/image_gen.py:create_image（m10；签名=00 号骨架登记 m10 第二行）

万相基址 = 环境变量("DASHSCOPE_BASE_URL", "https://dashscope.aliyuncs.com/api/v1")

async def create_image(prompt) -> dict | 任务失败:
    # ── 主干判断① prompt 契约（官方：input.prompt 必选、≤800 字符）──
    if not prompt or len(prompt) > 800:
        raise 生成输入不合(f"绘图 prompt 长度 {len(prompt)} 越界（1–800 字符）")
    # ── 数据变形① 创建任务：HTTP 仅支持异步（官方——缺异步头报
    #    "current user api does not support synchronous calls"）──
    创建响应 = await HTTP.post(f"{万相基址}/services/aigc/text2image/image-synthesis",
        headers={"Authorization": f"Bearer {环境变量('DASHSCOPE_API_KEY')}",
                 "X-DashScope-Async": "enable"},          # ★ 万相异步头（设计范围红线逐字）
        json={"model": "wanx-v1",
              "input": {"prompt": prompt},                # negative_prompt/style/size/n 可选档位
              "parameters": {"size": "1024*1024", "n": 1}})
    # ── 主干判断② 创建受理校验：万相对创建请求本身的拒绝（限流：任务下发 QPS 2、
    #    同时处理中任务 1——官方）——没有任务号就是没受理，不做假受理 ──
    任务号 = 创建响应.get("output", {}).get("task_id")      # 状态机：PENDING/RUNNING/SUCCEEDED/
    if 任务号 is None:                                      #   FAILED/CANCELED/UNKNOWN（官方）
        raise 任务失败(原因码="image_task_rejected", 任务号=None,
                       上游报文=创建响应, 阶段="创建")
    # ── 数据变形② 轮询任务详情：指数退避（首查 2 秒、逐次翻倍、10 秒封顶）+ 总超时兜底 ──
    起始 = 时钟.now(); 间隔 = 2
    while True:
        await asyncio.sleep(间隔)
        详情 = await HTTP.get(f"{万相基址}/tasks/{任务号}",
                              headers={"Authorization": f"Bearer {环境变量('DASHSCOPE_API_KEY')}"})
        状态 = 详情["output"]["task_status"]
        if 状态 == "SUCCEEDED":

    # 调用时点契约见下段声明，物理调用归设计稿 21

async def tts(text) -> AudioStream:
    # ── 主干判断① 定位声明（设计范围红线：无 SLA 仅原型定位写入设计稿）──
    #    edge-tts 封装微软 Edge 在线 TTS 服务：非微软官方 API、无 SLA（官方）——原型/演示可用，
    #    生产不可押注；生产迁移路径=国产云 TTS（官方文档国产化落地建议），届时只换本函数体，
    #    返回形状（音频流+字幕流）与调用方零改动
    # ── 主干判断② 文本非空 ──
    if not text.strip():
        raise 生成输入不合("待合成文本为空——零输入不出空音频")
    # ── 数据变形：流式合成音频 + 字幕原料 ──
    #    官方 CLI 形状（官方逐字）：edge-tts --voice zh-CN-XiaoxiaoNeural --text "…"
    #        --write-media out.mp3 --write-subtitles out.srt
    #    Python 侧按同形状调库（官方：模块可 import 使用、库内部走流式合成）
    通信器 = edge_tts.Communicate(text, voice="zh-CN-XiaoxiaoNeural")   # ★ 指定中文音色（设计范围红线）
    #   语速/音量/音高可调（rate/volume/pitch，负值必须写 -50% 形式——官方）；
    #   自定义 SSML 支持已移除（官方：服务端只接受单 voice 套单 prosody 结构）——调优边界
    音频流 = BytesIO(); 字幕原料 = []
    async for 块 in 通信器.stream():                       # 流式：边出边收，不整段驻留
        if 块["type"] == "audio":
            音频流.write(块["data"])                       # 音频字节块
        elif 块["type"] == "WordBoundary":
            字幕原料.append((块["offset"], 块["duration"], 块["text"]))   # 字幕时间戳原料
    音频流.seek(0)
    return {"音频": 音频流, "字幕": SRT 格式化(字幕原料), "音色": "zh-CN-XiaoxiaoNeural"}
    # AudioStream=音频流+字幕流双件（W-F10-4 要求音频与字幕双产物——流形状是两件的容器）；
    # 异常去向：合成服务不可达/合成失败 → 结构化异常上抛 → 设计稿 21 失败路径（同主干三形状）
```

## 关键判断的取舍记录

1. **PPT 走 python-pptx 代码生成（模板驱动），不让多模态模型直出二进制——可验证性优先。** 代码生成的产物是明确定义的数据结构：每页元素的位置/内容/样式都可程序化断言（标题非空、要点数与结论条数一致、数值表与分析关键数值逐格相等——月报里的数字错一位就是调度事故）；模型直出二进制无法逐项校验、也无法增量修订（改一处=整体重出）。模板文件把品牌版式锁死，分析 JSON→布局参数是确定性映射，主干零模型调用、零生成成本、产物可复现。备选（Slidev 类代码框架生成后再转 pptx）：多一道转换链路与渲染依赖，本系统产物就是 .pptx，无必要。换边条件：出现模板套不下的"自由设计"需求——那就是提议者-审核者进阶的入场券（主干二注记），代价是每页渲染审查与修订的模型成本。
2. **轮询用指数退避+总超时兜底，不用固定间隔，也不用 SDK 同步封装。** 万相任务耗时方差大（秒级到分钟级）：固定短间隔在长任务上烧请求配额（官方限流：任务下发 QPS 2、同时处理中任务 1——轮询请求同样占连接与额度），固定长间隔把"其实几秒就好"的任务拖出尾延迟；指数退避 2→4→8→10 封顶两头兼顾，总超时 10 分钟防"任务号活着但永远 RUNNING"挂死消费。不用 SDK `ImageSynthesis.call()` 同步封装：把"等"藏进库内=worker 消费节奏被库接管，任务号、轮询节奏、超时判据、失败边界全部不可控（设计稿 21 打捞重做需要 task_id 与明确的失败形状）；`async_call()+wait()` 的内部节奏同样不可配置。换边条件：万相官方提供任务完成回调/webhook 时，轮询整体换回调——create_image 的签名与返回形状不动，换的只是"等"的方式。
3. **转存义务"声明在本节、调用在设计稿 21"，不把 fetch_and_rearchive 塞进 create_image。** m10 卡面依赖只有 m2，"生成与交付分离"是 Seam 原判（生成器只产出产物候选，落桶/落库/通知全在 m15 统一经手）——转存义务"一处管"正是边界规则第 5 行的理由：让转存的时点与重试语义只有一个主人。备选（create_image 轮询成功后直调 m5）：要给 m10 添卡面没有的 m5 依赖（卡面登记所定，设计稿无权加），且"生成件知道桶规划"把交付政策漏进生成层。代价：转存链拆在三个设计稿（本节时点契约 + 设计稿 05 函数形状 + 设计稿 21 编排），走查时三段要拼着读——归属链在声明段写明。换边条件：出现不经 worker 的直连生成需求时，给 m10 卡补 m5 公开面依赖，再动函数体。
4. **asr 经网关统一入口调用，不直连 FunASR 服务端点。** 边界规则第 1 行（模型流量唯一出口是网关 :4000）对转写同样成立：转写也是模型流量，统一鉴权（同一 Virtual Key）、记账（网关 spend 覆盖，"转写不计账"就是账目破口）、路由（config.yaml 换部署别名零发版）三件白拿；官方文档明文 FunASR OpenAI 兼容服务"与 LiteLLM 网关的 OpenAI 形态天然对齐"。备选（业务侧直连 funasr-server）：三件都要自己重做一遍，且唯一出口纪律破口。代价：转写调用不经过 Gateway 类（m2 公开面只有 chat/embed 两签名），而是直用 OpenAI SDK 标准资源指向统一入口——形状一致性靠同一 base_url 与同一密钥注入保证。换边条件：FunASR 需要流式中间结果等超出 OpenAI 兼容面的能力时——网关 pass-through 或给 m2 卡扩面。

## 锚点对照精要

**docs/tech-stack/multimodal-api.md · python-pptx（quickstart 官方形状）**（卡 `m10` 锚点 K-C）

**原书第五章 §代码：通用 Agent 的元能力（代码驱动的多媒体生成）**（卡 `m10` 锚点）
书：代码生成是元能力（能创造其他能力的能力——精确、可执行、可组合），把文档创作从 GUI 视觉定位转化为对元素的精确程序化控制；Agent 写完代码并不知道实际渲染效果，故引入提议者-审核者机制——Proposer 生成代码、Reviewer 渲染截图用 Vision LLM 出结构化改进建议（含页码/问题类型/严重程度字段，不是模糊的"不好看"），迭代直到质量达标或最大轮数（Loop 工程两类显式终止条件）；双 Agent 分工的核心优势在上下文管理（Reviewer 只处理最新版渲染图、Proposer 只累积结构化文本反馈）。落点：主干一模板驱动=元能力"精确控制"的最小兑现（确定性映射、产物可逐项断言）；主干二注记按书机制定形（渲染导出用 LibreOffice headless 补 python-pptx 无渲染能力），修订上限一轮=预算上限类终止条件在本节取值，多轮化是换边条件。
