SPEC = ToolSpec(                                   # m8 契约五字段（设计稿 12 定形），本节照契约填
    name="identify_lithology",
    description="适用：用户问某口井某个井段'是什么岩性/砂岩还是泥岩/岩性怎么分布'——"
                "内部跑 FORCE 2020 复现的 XGBoost 测井岩性模型对曲线逐点判识，"
                "返回井段级压缩结论（主岩性占比+关键数值+图件引用），通常耗时数秒。"
                "不要用：查日产或生产数据（走 query_production_db）；"
                "报告里的地质结论文查证（走问答 Agent 报告库，模型只认测井曲线）；"
                "未来产量外推（走 forecast_production）；"
                "没说清井段的泛问（先追问井段，凭空选段结论无意义）。",
    parameters={"type": "object",
                "properties": {
                    "well_id": {"type": "string",
                                "description": "井号，如 F-1 / N-2（须在井档案内）"},
                    "depth_range": {"type": "array", "items": {"type": "number"},
                                    "minItems": 2, "maxItems": 2,
                                    "description": "井段 [顶深m, 底深m]，如 [2500, 2600]"}},
                "required": ["well_id", "depth_range"]},
    permission_level="readonly",                   # 🛡️ 只读：模型推理无外部副作用（W-F9-2 同级放行的依据）

async def 识别岩性(well_id, depth_range) -> LithologyResult:
    # ── 主干判断① 入参校验（语义层，schema 层已由设计稿 12 执行器第③步拦过形状）──
    顶深, 底深 = depth_range
    if 顶深 >= 底深:
        raise 参数不合(f"井段顶深 {顶深} 应小于底深 {底深}")
    井档案 = await 读井档案(well_id)                # IO 边界 await：读本目录离线 LAS 档案（取舍 2）
    if 井档案 is None:
        # 查无此井 → 参数不合（带可用井清单，模型可向用户说明或改问——主干五全程走此支线）
        raise 参数不合(f"井号 {well_id} 不在井档案中；当前档案可用井号：{档案井号清单()}")
    if 井段超出(井档案.井深范围, depth_range):
        raise 参数不合(f"井段 {depth_range} 超出 {well_id} 井档案范围 {井档案.井深范围}")
    # ── 主干判断② 演示模式分支（替身保留声明；设计稿 00 stub #11 的保留侧）──
    if 演示模式开关开:
        return FakeDomainModel.固定岩性结论(well_id, depth_range)   # 形状与真身结论同构
    # ── 数据变形① 裁段取曲线：全井 2600 点 → 本井段 101 点（曲线只在本函数体内流动）──
    井段曲线 = 井档案.裁段(depth_range)             # 每点 {深度, GR, RT, DEN, NPHI, CAL, SP}
    # ── 数据变形② 模型推理：内部跑既有 FORCE 2020 预训练模型文件（XGBoost），只推理不训练；
    #    算法原理不展开（设计范围"明确不讲的"）；首次调用装载进程内缓存（IO await）──
    岩性标签序列 = await 岩性模型.预测(特征(井段曲线))   # 每深度点一个岩性标签 + 概率
    # ── 数据变形③ 结论压缩：逐点标签 → 段级结论（主干三展开 2600 点口径的代价推演）──
    return {"well_id": well_id, "depth_range": depth_range,
            "主结论": 主岩性判读(岩性标签序列),       # 如"砂岩为主，中部夹泥岩互层"
            "岩性占比": 占比统计(岩性标签序列),       # 如 {"砂岩": 0.72, "泥岩": 0.21, "粉砂岩": 0.07}
            "关键数值": {"井段净厚m": 底深-顶深, "砂岩段累计m": 累计厚度(岩性标签序列),
                        "岩性切换次数": 切换数(岩性标签序列), "最低置信深度m": 最弱点(岩性标签序列)},
            "置信度": 均值概率(岩性标签序列),          # 判读可信度，模型自己给出
            "可视化引用": await 落柱状图件(岩性标签序列, well_id, depth_range),
            # ↑ 图件落盘为工件、只回引用键（图不进上下文——同产物"只存引用不存正文"纪律）
            "模型版本": 岩性模型.版本}
    # ★ 原始曲线（井段曲线变量）在此函数返回后即弃——不 return、不进上下文（W-F9-1 期望的机制位）

async def 预测产量(well_id, days) -> ForecastResult:
    # ── 主干判断① 入参校验：井号存在性（时序档案，与岩性井档案同目录离线件）──

        return FakeDomainModel.固定预测结论(well_id, days)

    # ── 数据变形② 模型推理：既有 LSTM 预训练模型文件，只推理不训练（10–30 秒，
    #    全程 await——可取消点恒在，客户端中途断开即中止，锚设计稿 01 捣乱 B 语义）──
    预测序列 = await 产量模型.外推(历史窗, 天数=days)

                        "区间低": 分位(预测序列, 0.1), "区间高": 分位(预测序列, 0.9)},
            "序列摘要": 隔日抽样(预测序列),           # 摘要级数字序列（逐日全量同样不进上下文）
            "可视化引用": await 落预测图件(预测序列, 历史窗, well_id, days),
            "模型版本": 产量模型.版本}
    # 异常去向：模型文件读失败/推理超时 → 下游异常上抛 → 设计稿 12 执行器三分法分支③结构化错误码

## 关键判断的取舍记录

1. **压缩在工具边界内完成（返回即结论），不做"全量返回+主循环压缩/截断"。** 模型要的是判识结论，不是数据本身；主循环侧压缩发生在 token 已计费、窗口已占用之后（捣乱 A 推演一），且"让模型自己从 2600 行数字里数占比"是把感知工具的活转嫁给 LLM——贵且不稳。备选一是 read 类工具的 offset/limit+截断标示（原书 §4 读取类做法）：适用"模型需要自选读哪段"的场景，岩性问句自带井段、无自选需求，截断只制造"看到全部"的错觉；备选二是"工具返回全量、提示词教模型总结"：压缩质量取决于提示词与模型心情，且每轮房租照付。换边条件：出现"模型需要自行浏览曲线细节再决定看哪段"的需求（探针式分析）时，加一个带 offset/limit 的曲线读取工具——与压缩结论工具并存，各管一种粒度，不替换。
2. **领域数据走自带离线井档案，不在线查生产库——m9 卡面依赖只有 m8。** 井号存在性校验与模型输入（LAS 曲线/时序窗）都从工具目录下的离线数据档案读，零 import m3：换真身不牵连持久化层，领域工具保持"自带干粮"的插件形态（工具体系 Seam：具体工具只经注册进入体系）。备选：经 m3 Repository 公开面读生产库（边界规则第 7 行允许面）——能拿到最新实绩，但给 m9 添了一条卡面没有的模块依赖，卡面归属已由登记裁定，本设计稿无权加（接口/依赖变更只能改卡）。换边条件：业务要求预测基于"实时最新生产数据"或井档案要运营化增删时，扩卡给 m9 卡补 m3 公开面依赖，再改读数站——工具 SPEC 与压缩结论形状不受影响。
3. **查无此井并入三分法分支①（param_invalid），不新增错误码枚举。** 设计稿 12 已把 Executor 的错误容器定形为三分法（参数错/权限拒/下游错），"井不在档案"本质是入参语义错（参数错族），语义信息全部放 message+hint（带可用井清单——它才是模型能拿去用的部分）。备选：给 m8 加 `well_not_found` 之类领域错误码——错误码枚举是 m8 机制面，设计稿 13 无权改 m8 卡；且为每个领域语义开错误码会让枚举无限膨胀。换边条件：若需要按"查无此井"做专门 UI/统计口径，扩卡在 m8 卡把错误码定成开放枚举（领域可注册），届时本工具只改错误码字符串，回灌路径不动。
4. **30 秒长任务走同步等待+执行中帧，不进异步任务队列（问智模式）；协议扩第六型 tool_progress 保持全帧同构。** 备选一是复用 `token` 帧装进度：token 的语义是"答案文本增量"，前端要把进度当正文过滤特判，破坏设计稿 01"全帧单一路径解析"的取舍；备选二是入 m4 任务队列异步化（问智受理-通知模式）：30 秒在网关 request_timeout 同量级，同步等待可承受，拉一条队列+受理帧+完成通知的链路（设计稿 19/21 那套）代价远大于收益；备选三是只靠 keep-alive ping：ping 是注释行，保连接不保 UX。所以：执行中帧 = 新帧型（语义清晰、可编程、{event,data} 同构），m9 侧义务只有两件——description 标注执行代价、func 全程 async 保可取消。换边条件：预测时长到分钟级（更大模型/更长窗）时，同步等待挤压连接预算，重新评估改走 m4 队列——届时 SPEC 的参数与压缩结论形状不变，只有调用方从 Executor 同步等待换成受理-通知。

## 锚点对照精要

**原书 §4 · 感知工具（返回信息压缩）**（本节回查锚点，任务指定）
书：感知工具常面临返回信息量远超 Agent 处理能力，直接塞入上下文既耗尽窗口又让关键内容淹没在噪声中，通用做法是工具层面做上下文感知压缩；搜索类返回结构化候选而非全文，读取类用 offset/limit 且截断必须标示（静默截断危险）；只读性带来缓存与并行两项工程红利。落点：压缩做在工具边界内（return 即压缩结论，捣乱 A 是反面推演）；截断式返回不采用（问句自带井段，无自选读取需求）；"缓存/并行红利"本节未接——结果缓存需 m4 Cache、m9 卡面零 m4 依赖，记为落点差异与潜在演进项。

**原书 §10 · 多 Agent 何时真正优于单 Agent（工具反馈=新信息）**（卡 `m9` 锚点 K-D 落地实例）
书：多 Agent/外部环节是否值得，判据只有一条——协作过程是否引入单个 Agent 生成时无法获得的新信息；表 10-1 里"审核者使用外部工具验证事实"属工具反馈=新信息、效果显著提升，而同一模型自我审查不引入新信息。落点：两个领域模型的推理结论（测井逐点判识、时序外推）正是主循环 LLM 生成时不可能自产的新信息——感知工具的正当性由此成立；捣乱 B 里选错工具查回的"真实列名"同样是新信息，模型据此自纠换工具，闭环仍是"执行-验证-反馈"。

**石油AI智能体项目学习计划（FORCE 2020 / Volve 背景）**（卡 `m9` 锚点 K-D 落地实例）
计划：FORCE 2020 测井竞赛（118 口井公开数据+冠军队 XGBoost+物理特征工程方案）是岩性识别唯一主力参考；产量预测取 PetroAI-Suite 思路+Volve 真实生产时序的 LSTM，封装形状即 `forecast_production(well_id, days)`；用户故事原句"帮我分析 F-1 井 2500-2600m 的岩性"。落点：本节两工具是该方案的"只推理不训练"封装——模型文件为预训练产物，训练脚本不在本系统（设计范围"不做什么"）；问句、井号、井段均沿用该计划原始设定。

**原书 §4 · 工具描述的艺术（选型半程的依据）**（配合主干四；卡 `m8` 锚点在本节的借用面）
书：描述核心是"什么时候用"而非"能做什么"，清晰列出边界与反例比描述能力更重要；选错工具时优先检查描述而非怀疑模型；应标注执行代价。落点：两 SPEC 的"适用："一句逐词对准业务问句形态、"不要用："段互指导友工具点名；forecast_production 标注"通常耗时 10–30 秒"即执行代价条目；捣乱 B 的修复顺序（先改描述）直接以此条为据。
