### 主干一：注册契约——ToolSpec 五字段与 Registry.register（W-F8-1 上半）

```text
伪代码 — app/agents/tools/registry.py:ToolSpec / Registry（m8 机制真身；签名=设计稿 00 登记表 m8 前四行）

class ToolSpec:                                   # 数据形状（卡面类型行）：五字段缺一不可
    name: str                # 注册名，全小写 snake_case（^[a-z][a-z0-9_]{1,63}$）——大小写敏感声明的机械面
    description: str         # 描述契约：必须含"适用："与"不要用："两段（什么时候用+反例，注册闸做结构检查）
    parameters: dict         # 参数 JSON Schema（本节使用子集：type/properties/required/items/
                             #   minItems/maxItems/minimum/maximum/enum/default）
    permission_level: str    # 权限级 ∈ {"readonly", "invoke"}：只读（对内无副作用）/ 外部调用（有外部副作用）
    func: callable           # 真身执行体；IO 边界一律 async（锚 modules.md 伪代码通用约定）

class Registry:
    def __init__(self):
        self._tools = {}                            # name → ToolSpec，唯一索引就是精确 name

    def register(self, spec) -> None:               # 注册主干：规格校验 → 重名拒 → 入表
        # ── 主干判断① 规格校验：五字段逐个过闸，任一不过拒注册（快速失败，不带病入表）──
        if not 合法标识符(spec.name):                # 全小写 snake_case，防同名变体混入
            raise 拒绝注册(f"工具名 {spec.name} 不合契约：全小写 snake_case")
        if not ("适用：" in spec.description and "不要用：" in spec.description):
            raise 拒绝注册(f"{spec.name} 的 description 缺'适用：/不要用：'两段——"
                           "描述契约是模型选型的唯一依据（取舍 1），不合格不上线")
        if not 合法参数模式(spec.parameters):        # object 型 JSON Schema、properties/required 形状自洽
            raise 拒绝注册(f"{spec.name} 的 parameters 不是合法 JSON Schema")
        if spec.permission_level not in ("readonly", "invoke"):
            raise 拒绝注册(f"{spec.name} 权限级必须显式声明为 readonly / invoke")
        if not callable(spec.func):
            raise 拒绝注册(f"{spec.name} 缺执行体")
        # ── 主干判断② 重名拒：同名已存在 → 拒绝，不静默覆盖（防同名工具遮蔽，锚 原书 §4）──
        if spec.name in self._tools:
            raise 拒绝注册(f"工具 {spec.name} 已注册——重复注册被拒；换名或先下线旧版")
        # ── 主干判断③ 入表 ──
        self._tools[spec.name] = spec

    def lookup(self, name) -> ToolSpec | None:      # 精确匹配；查无此工具是执行器的"结构化缺失"输入
        return self._tools.get(name)                # 大小写敏感："Query_Production_DB" ≠ "query_production_db"
```

### 主干二：工具清单注入上下文——按角色过滤可见性（W-F8-1 下半 + W-F8-4 前半）

```text
伪代码 — app/agents/tools/registry.py:Registry.tools_for 与 app/agents/tools/permission.py:Permission（m8）

class Permission:                                 # 机制与政策分离：映射表是政策，Permission 是机制
    # 角色 → 允许权限级集合（政策表，随角色账号配置走；user 以数据形状传入，零 import m1）
    ROLE_TOOL_LEVELS = {
        "reservoir_engineer":  {"readonly", "invoke"},   # 老王：油藏工程师账号
        "geologist_readonly":  {"readonly"},             # 小李：地质工程师只读档（捣乱 A 主角）
        "observer":            {"readonly"},             # 主任：观摩账号
    }

    def filter_tools(self, user, tools) -> VisibleTools:      # 🛡️ 按角色过滤工具可见性
        允许级 = self.ROLE_TOOL_LEVELS[user.role]
        return [spec for spec in tools if spec.permission_level in 允许级]
        # 可见性闸：无权工具根本不进请求——模型"看不见"就不会选它（捣乱输入 A 第①道）

    def assert_allowed(self, user, spec) -> 拒绝理由 | None:   # 🛡️ 执行前权限闸（主干三第④步调）
        允许级 = self.ROLE_TOOL_LEVELS[user.role]
        if spec.permission_level in 允许级:
            return None
        return f"当前账号角色 {user.role} 为只读权限，不允许调用 {spec.permission_level} 级工具"
        # 直连兜底闸：模型幻觉出看不见的工具名时，执行前再拦一次（两道闸各防一条绕过路径）

class Registry:
    MAX_TOOLS_PER_TURN = 20                        # 官方软建议每轮起始 ≤20（锚 OpenAI 官方文档）

    def tools_for(self, user) -> tools 数组:        # 主循环每轮请求前调（设计稿 17 接线）
        # ── 数据变形① 全量 → 按角色过滤 ──
        visible = self._permission.filter_tools(user, self._tools.values())
        # ── 数据变形② ToolSpec → OpenAI tools 数组官方形状（OpenAI 官方文档：type 目前只支持 "function"）──
        tools_array = [{"type": "function",
                        "function": {"name": spec.name,
                                     "description": spec.description,   # 描述契约原文随声明出
                                     "parameters": spec.parameters}}
                       for spec in visible]
        # ── 主干判断③ 数量闸：超限显式失败，不静默截断 ──
        if len(tools_array) > self.MAX_TOOLS_PER_TURN:
            raise ToolBudgetExceeded(                  # 上线/装配期就该修工具集——静默截断会让模型
                f"角色 {user.role} 可见工具 {len(tools_array)} 条 > 20，"   # 看不见被砍工具而选错/编造
                "在装配期收敛工具集或按场景分组")
        return tools_array
```

```text

class Executor:
    def __init__(self, registry, permission):       # m8 内部件注入；下游工具一律按注册名解析
        self._registry, self._permission = registry, permission

    async def call(self, tool_call, user) -> ToolResult:
        # ToolResult 数据形状（本节定形）：{call_id, name, ok, content, error_code?}
        #   call_id 钉死配对（并行多调用不错配的根）；content 统一 JSON 文本（成功与错误同形状）
        call_id = tool_call.id                      # 官方形状：tool_calls[] 元素的 id（"call_xxx"）
        name = tool_call.function.name
        # ── 第①步 查表：查无此工具 = 结构化缺失，也是"可纠正错误"的一种 ──
        spec = self._registry.lookup(name)
        if spec is None:
            return ToolResult(call_id, name, ok=False, error_code="unknown_tool",
                content=错误文本({"error_code": "unknown_tool",
                                  "message": f"工具 {name} 未注册",
                                  "hint": "工具名大小写敏感、全小写 snake_case，请在可见工具清单内选择"}))
        # ── 第②步 参数解析：arguments 是 JSON 字符串（官方形状），解析失败工具零执行 ──
        try:
            args = json.loads(tool_call.function.arguments)
        except JSON解析失败 as e:
            return ToolResult(call_id, name, ok=False, error_code="param_invalid",
                content=错误文本({"error_code": "param_invalid",
                                  "message": f"arguments 不是合法 JSON：{e.位置}",
                                  "hint": "重新输出完整的 JSON 对象，不要截断或加注释"}))
        # ── 第③步 schema 校验：形状不合即回灌，禁静默转换/禁补默认值（参数保真禁令，取舍 3）──
        issues = self._validate(args, spec.parameters)   # 逐条判：类型/必填/items 长度/数值范围/枚举
        if issues:
            return ToolResult(call_id, name, ok=False, error_code="param_invalid",
                content=错误文本({"error_code": "param_invalid",
                                  "message": [f"参数 {i.参数名} 实为 {i.实际形状}，"
                                              f"期望 {i.期望形状}" for i in issues],
                                  "hint": "按 parameters 的 JSON Schema 修正后重试"}))
        # ── 第④步 权限复核：可见性闸看不见 ≠ 调不到——直连兜底闸在此（捣乱 A t2）──
        拒绝理由 = self._permission.assert_allowed(user, spec)
        if 拒绝理由:
            return ToolResult(call_id, name, ok=False, error_code="permission_denied",
                content=拒绝理由)                    # 理由即回灌内容：入轨迹，模型可向用户转述
        # ── 第⑤步 执行：工具内异常只分两族，全部折成结构化 ToolResult，不上抛裸异常 ──
        try:
            result = await spec.func(**args)         # IO 边界 await（锚 modules.md 伪代码通用约定）
        except 参数不合 as e:                         # 工具侧业务校验拒（非 SELECT / 条件必填缺失）
            return ToolResult(call_id, name, ok=False, error_code="param_invalid",
                content=错误文本({"error_code": "param_invalid",
                                  "message": str(e),
                                  "hint": "按工具描述修正参数后重试"}))   # → 三分法分支①（可纠正）
        except 下游异常 as e:
            return ToolResult(call_id, name, ok=False, error_code=归类错误码(name, e),
                content=错误文本({"error_code": 归类错误码(name, e),
                                  "message": 可解释消息(e),       # "30 秒未响应，调用未确认成功"
                                  "hint": 重试指引(e)}))          # 副作用未确认→提示先查询再重试
        return ToolResult(call_id, name, ok=True, content=转JSON文本(result))

    def _validate(self, args, schema) -> issues:     # 本节 JSON Schema 子集的判别器（不引全量 Draft 依赖）
        issues = []
        for 必填名 in schema.get("required", []):     # required：缺参即错
            if 必填名 not in args: issues.append(问题(必填名, "缺失", schema 期望))
        for 名, 值 in args.items():                   # 逐参数：type / enum / items 长度 / minimum/maximum
            期望 = schema["properties"].get(名)
            if 期望 is None: issues.append(问题(名, 值形状, "未声明的参数（幻觉参数）"))   # 幻觉参数在此落网
            elif not 形状匹配(值, 期望): issues.append(问题(名, 值形状, 期望形状))
        return issues                                 # 有 issues = 校验失败；零隐式转换（取舍 3）
```

## 关键判断的取舍记录

1. **把"什么时候用+反例"定成可校验的描述契约（"适用：/不要用："两段结构检查入注册闸），不做自由文本。** 模型选工具时唯一能看见的就是 description——它看不到实现代码；书的消融证据是附 1–5 个调用示例可使选型准确率约 72%→90%、"清晰列出边界与反例比描述能力更重要"，选错工具时优先怀疑描述而不是模型（锚 原书 §4）。所以描述质量直接决定选型准确率，值得让"不合格描述"在注册闸就上不了线。备选：自由文本+上线前人工评审。不选：评审管不住质量下限，也没法在"新工具上线"（W-F8-1）时机械判定契约齐不齐。本节做结构检查（两段在不在），语义质量仍靠评审——闸保下限，人保上限。换边条件：若后续引入调用示例字段（examples），结构检查扩展为"示例 ≤5 条"的形状校验，两段检查不撤。
2. **校验分三层：schema 管形状、description 管语义、代码管安全与业务不变量；不启用 strict 模式。** parameters（JSON Schema）判类型/必填/枚举/范围——机器可判、失败可机械表述；description 判"该不该用这个工具"——只有模型能判；安全不变量（invoke_api 白名单、row_limit 收口、SELECT 末闸）写在 func 里——不信任模型也不信任 schema 能表达全部约束。备选一是全塞 schema：OpenAI 官方文档的 `strict: true` 要求 additionalProperties:false 且所有 properties 列入 required，可选字段得扭曲成 `type:["string","null"]`——表达代价高，且 strict 只保"合 schema"，保不住白名单这类语义约束；备选二是全靠代码校验：错误发现得太晚，模型没有 schema 指引会反复错在同一种形状上。换边条件：当模型端 structured outputs 成熟且网关全链路支持 strict 时，形状层可切 strict，代码层安全约束永远保留。
3. **参数保真禁令：校验失败即回灌重试，执行器禁止静默转换入参、禁止补 schema 未声明的默认值。** 备选：执行器"友好地"把 "2500-2600米" 解析成 [2500,2600] 再执行（捣乱输入 B 的反面）。不选：书里的反模式实证——Cursor 把弯引号静默转直引号、IDE 的 bash 工具擅自附加 git 参数，都属于"静默输入转换/静默参数注入"，本质是模型感知的世界与工具操作的世界之间出现系统性偏差（锚 原书 §4）；静默转换这次对、换个写法就错，且模型从成功里学不到正确格式，错误永远修不掉。回灌重试把纠错责任还给模型——它有 schema、有错误消息，一轮就能改对。换边条件：仅对"无损且可逆"的数值宽容（如 schema 用 number 同收整数与浮点）可以在 schema 层放宽——那仍是声明过的契约，不是执行器偷偷转换。schema 声明过的 default（row_limit=50）不算补默认值：默认值是契约的一部分。
4. **权限双闸（可见性过滤 + 执行前复核）与错误即数据（ToolResult 统一容器，三分法是数据不是异常流）。** 双闸各防一条绕过路径：只做 filter_tools——模型可能幻觉出数组里没有的工具名直连（OpenAI 官方文档明示 arguments 会幻觉，工具名同理），assert_allowed 兜底；只做 assert_allowed——无权工具留在清单里既挤占每轮 ≤20 的预算，又诱导模型反复选择必被拒的工具。错误侧：Chat Completions 协议要求每个 tool_call_id 都有一条 role:"tool" 应答——"抛异常"在协议层无处可抛（异常承载不了 tool_call_id），所以 Executor 的三类错误全部折成 ToolResult 数据回灌，参数错带重试指引、权限拒不带指引（不给绕权提示）、下游错带副作用提示。换边条件：若出现"拒绝次数超阈值"的滥用模式，在主循环（设计稿 17）加熔断判据——执行器形状不动。

## 锚点对照精要

**docs/tech-stack/openai-embedding.md · tools 字段与工具声明形状**（卡 `m8` 锚点 K-D）
文：tools 数组元素形状 `{"type":"function","function":{"name","description","parameters"(JSON Schema),"strict"?}}`，type 目前只支持 "function"；函数定义以注入 system 消息方式计入输入 token 并计费，官方软建议每轮起始可用函数 ≤20。落点：主干二 tools_for 的导出形状逐字段按此，MAX_TOOLS_PER_TURN=20 闸与"超限显式失败"的出处即这条计费+软建议。

**docs/tech-stack/openai-embedding.md · 五步循环与回灌配对**（卡 `m8` 锚点 K-D）
文：官方五步——带 tools 请求→收到 tool call→应用侧执行（API 本身不执行任何函数）→以 role:"tool" 消息回灌、tool_call_id 必须对应上一条 assistant 消息里的某个 id→二次请求得终答；含 tool_calls 的 assistant 消息必须原样入历史；新模型支持一轮并行多调用，代码必须按 N 个处理。落点：主干三回灌契约块逐条锁死（原样入历史 / N 对 N / call_id 钉死配对）；主循环 while 半程归设计稿 17。

**docs/tech-stack/openai-embedding.md · arguments 必须校验与默认非 strict**（卡 `m8` 锚点 K-D）
文：`arguments` 是 JSON 字符串而非对象，官方警告模型可能生成非法 JSON 或幻觉参数、必须在代码里校验；`strict: true` 走 structured outputs 可保参数严格合 schema，但 Chat Completions 默认非 strict。落点：主干三第②③步把"必须校验"落成两道机械闸；取舍 2 的"不启用 strict、校验留在应用侧"即以"默认非 strict"为协议面依据。

**原书 §4 · 工具描述的艺术**（卡 `m8` 锚点 K-D 第一项）
书：描述核心是"什么时候用"而非"能做什么"，清晰列出边界与反例比描述能力更重要；参数用具体例子、返回值写清结构、标注执行代价；附 1–5 个调用示例可使准确率约 72%→90%；Agent 选错工具时优先检查描述而非怀疑模型。落点：ToolSpec.description 契约化为"适用：/不要用："两段且注册闸结构检查（取舍 1）；所有 description 都写反例段；错误回灌消息带期望形状，让"检查描述"有机械入口。

**原书 §4 · 参数传递的保真性**（卡 `m8` 锚点 K-D 第一项）
书：反模式有二——"静默输入转换"（Cursor 弯引号转直引号）与"静默参数注入"（bash 工具擅自附加 git 参数）；核心原则是模型感知的世界与工具操作的世界之间不能有系统性偏差。落点：捣乱输入 B 的反面（执行器好心解析字符串井段）即"静默输入转换"在本系统的化身；取舍 3 的禁令与回灌重试是落点，schema 声明的 default 是契约、不算注入。

**原书 §4 · 执行工具的多层防护与执行-验证-反馈闭环**（卡 `m8` 锚点 K-D 第一项）
书：执行工具错误代价极高，要在能力开放与安全约束间平衡，防护分层——输入验证+权限控制；提议者-审核者模式下"拒绝理由作为工具结果入轨迹"；执行-验证-反馈闭环让工具反馈成为模型的新信息；幂等性靠唯一标识去重、先查询后变更。落点：五步闭环=该闭环的最小形态（验证=schema+权限两道，反馈=ToolResult 回灌）；捣乱输入 A 的拒绝理由入轨迹与书同构；invoke_api 回执号+先查询后重试对应幂等三招。

**原书 §4 · ACI 与工具五分类**（卡 `m8` 锚点 K-D 第一项）
书：ACI 对标 HCI——工具应面向 Agent 的目标而非底层 API（三条原则：能力用什么形式表达、工具怎么描述、参数如何忠实传递）；工具五分类=感知/执行/协作/事件触发/用户沟通。落点：ToolSpec 五字段即三原则的机械面（parameters=形式、description=描述、校验闸=保真）；内置三工具各占一类样板——query_production_db=执行（只读）、navigate_page=用户沟通（界面指路）、invoke_api=执行（外部动作），设计稿 13 领域工具是感知类（返回压缩结论）。
